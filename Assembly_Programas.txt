Programa 1: Sequência de Fibonacci (Laço Iterativo) 



// --- ENTRADA ---

[00] IN R1              // Lê as chaves (n) e salva em R1.

[01] BEQ R1, R0, +10    // Se n == 0, pula lá para o final (linha 12).



// --- INICIALIZAÇÃO ---

[02] ADDI R2, R0, 0     // R2 (a) = 0

[03] ADDI R3, R0, 1     // R3 (b) = 1

[04] ADDI R4, R0, 2     // R4 (i) = 2



// --- INÍCIO DO LOOP ---

[05] SLT R5, R1, R4     // LOOP: Se n < i, R5 = 1. Senão, R5 = 0.

[06] BNE R5, R0, +6     // Se R5 != 0 (i > n), pula para o fim (linha 13).

[07] ADD R5, R2, R3     // temp = a + b

[08] ADDI R2, R3, 0     // a = b

[09] ADDI R3, R5, 0     // b = temp

[10] ADDI R4, R4, 1     // i = i + 1

[11] JUMP 5             // Volta incondicionalmente para avaliar o loop.



// --- CASOS BASES E SAÍDA ---

[12] ADDI R3, R0, 0     // END_ZERO: Se n = 0, resposta recebe 0.

[13] OUT R3             // Mostra o resultado nos displays HEX.

[14] HLT                // Interrompe o processamento.

Programa 2: Fatorial Recursivo 



// --- GRAVAÇÃO E LEITURA NA RAM ---

[00] IN R1               // Lê as chaves para n (ex: 5).

[01] SW_ABS R1, 15       // Endereçamento Absoluto: Memória[15] = R1.

[02] ADDI R5, R0, 10     // R5 recebe 10 (usaremos como Base).

[03] LW R1, 5(R5)        // Endereçamento Base+Offset: R1 = Memória[R5 + 5]. Lê o valor de volta!



// --- INÍCIO DA RECURSÃO ---

[04] JAL 7               // Pula para a subrotina FAT e salva o PC+1 no R31.

[05] OUT R2              // O resultado do Fatorial volta em R2. Mostra no display!

[06] HLT                 // Interrompe o processamento.



// --- SUBROTINA RECURSIVA: FAT — //

[07] BEQ R1, R0, +8      // Se n == 0, pula para o caso base na linha 16.

[08] PUSH R31            // Salva o endereço de retorno do chamador atual na pilha.

[09] PUSH R1             // Salva o 'n' local na pilha para usar depois do retorno.

[10] ADDI R1, R1, -1     // Decrementa: n = n - 1

[11] JAL 7               // Chamada recursiva para FAT.

[12] POP R1              // Restaura o 'n' que foi guardado na linha 09.

[13] POP R31             // Restaura o endereço de retorno do chamador (Linha 08).

[14] MUL R2, R1, R2      // R2 = n * fatorial(n-1). Usa DSP Nativo.

[15] JR R31              // Retorna a execução para o chamador.



// --- CASO BASE DO FATORIAL ---

[16] ADDI R2, R0, 1      // Se n = 0, R2 = 1.

[17] JR R31              // Retorna.
